# 1. Вложенные циклы. Примеры использования
Короче, вложенный цикл это когда один цикл сидит внутри другого. Работает так: внешний цикл делает один шаг, а внутренний за это время пробегает весь свой путь от начала до конца. И так по кругу, пока внешний не закончится.
Где это юзают? Ну, например, когда надо пройтись по двумерному массиву — внешний цикл идёт по строкам, а внутренний по столбцам. Ещё таблицу умножения так выводят. Или фигурки всякие в консоли рисуют звёздочками. Больше как-то на ум ничего не приходит:::::(((.
# 2. Локальные и глобальные переменные. Время жизни и область видимости
Локальная переменная — это та, которую объявил внутри метода или какого-то блока в фигурных скобках. Её видно только внутри этого блока, а как блок закончился — всё, она из памяти удаляется. И вот что важно в C#: локальной переменной надо обязательно сразу дать значение, иначе компилятор ругнётся и не даст запустить. А глобальных переменных в чистом виде в C# вообще нет, потому что всё должно лежать внутри класса. Их роль играют поля класса. И вот прикол: поля, в отличие от локальных, сами получают значения по умолчанию — ноль для чисел, false для булевых, null для ссылочных типов. Ничего вручную задавать не надо. Область видимости — это просто то место в коде, где переменную можно использовать. Ну то есть пока она существует в памяти, там её и видно.

   ```C#
class Program

        double b = double.Parse(Console.ReadLine()); // Считывает сторону b

        // Главное правило: сумма двух ЛЮБЫХ сторон должна быть строго больше третьей
        if (a + b > c && a + c > b && b + c > a)
        {
            Console.WriteLine("Треугольник построить можно!"); 

            // проверяем тип треугольника с помощью вложенных условий
            if (a == b && b == c) // Если все три стороны равны
            {
                Console.WriteLine("Тип: Равносторонний");6
            }
            else if (a == b || a == c || b == c) 
            {
                Console.WriteLine("Тип: Равнобедренный"); 
            }
            // Проверка на прямоугольный (теорема Пифагора: квадрат гипотенузы равен сумме квадратов катетов)
            else if ((a * a + b * b == c * c) || (a * a + c * c == b * b) || (b * b + c * c == a * a))
            {
                Console.WriteLine("Тип: Прямоугольный"); // Если подошло хоть под одну формулу то прямоугольный
            }
            else 
            {
                Console.WriteLine("Тип: Обычный (разносторонний)"); // // Если ни одно условие выше не сработало, значит все стороны разные
            }
        }
        else // Если самое первое условие (про сумму сторон) не прошло
        {
            Console.WriteLine("Треугольник построить нельзя :("); // // Если самое первое условие (про сумму сторон) не прошло, значит что нельзя
        }

        Console.ReadLine();
    }
}
   ```

2. Циклы
```c#
using System; 

class Program 
{
    static void Main() // Точка входа
    {
        Console.WriteLine("Числа Армстронга от 100 до 999:"); // Выводим заголовок

        // Запускаем цикл от 100 до 999 (включительно)
        for (int i = 100; i <= 999; i++)
        {
            // Достаем цифры из трехзначного числа i с помощью математики
            int sotni = i / 100;          // Деление на 100 отбрасывает дробную часть 
            int desyatki = (i / 10) % 10; // Делим на 10 (получаем 15),
            int edinicy = i % 10;         // Просто берем остаток от деления на 10, 

            // Считаем сумму кубов этих цифр
            int summaKubov = (sotni * sotni * sotni) +
                             (desyatki * desyatki * desyatki) +
                             (edinicy * edinicy * edinicy);

            // Проверяем: если сумма кубов равна самому числу, то это число Армстронга
            if (summaKubov == i)
            {
                Console.WriteLine(i); 
            }
        }

        Console.ReadLine(); 
    }
}
```

3. Массив
```c#
using System;

class Program 
{
    static void Main() 
    {
        // Создаем массив с числами для проверки (можно любые)
        int[] array = { 4, 8, 15, 16, 23, 42 };
        int n = array.Length; // Запоминаем длину массива 

        // 1. Сначала найдем сумму всех элементов массива
        int summa = 0; 
        for (int i = 0; i < n; i++) 
        {
            summa += array[i]; // Прибавляем к копилке текущий элемент массива
        }

        // 2. Считаем среднее арифметическое (сумму делим на количество)
        double srednee = (double)summa / n;
        Console.WriteLine("Среднее арифметическое: " + srednee); // среднее

        // 3. Ищем, сколько элементов строго больше среднего
        int count = 0; 
        for (int i = 0; i < n; i++) // Опять бежим по всему массиву
        {
            if (array[i] > srednee) // Если элемент больше, чем среднее
            {
                count++; 
            }
        }

        Console.WriteLine("Количество элементов больше среднего: " + count); 

        Console.ReadLine(); 
    }
}
```
